4. Almacenamiento de Procedimientos (Stored Procedures - 2 SP)

Desarrolle e implemente 2 Stored Procedures requeridos para la lógica del sistema agroindustrial con control de concurrencia y validaciones de stock.

4.1 Stored Procedure 01: Registrar Venta con Control de Stock Atómico (RN-VNT-01)

Regla RN-VNT-01: Antes de registrar el ticket y detalle, se bloquea el registro del producto con FOR UPDATE. Si el stock en catálogo o lote no abastece la cantidad solicitada, lanza la excepción personalizada -20010 evitando sobreventas.

CREATE OR REPLACE PROCEDURE SP_REGISTRAR_VENTA_ITEM (
    p_cliente_nombre IN VARCHAR2,
    p_metodo_pago    IN VARCHAR2,
    p_id_producto    IN NUMBER,
    p_id_lote        IN NUMBER DEFAULT NULL,
    p_cantidad       IN NUMBER,
    p_numero_ticket  OUT VARCHAR2,
    p_total_cobrado  OUT NUMBER
) AS
    v_stock_actual   NUMBER(10);
    v_precio_prod    NUMBER(10, 2);
    v_nombre_prod    VARCHAR2(120);
    v_subtotal_item  NUMBER(12, 2);
    v_igv            NUMBER(12, 2);
    v_base_neta      NUMBER(12, 2);
    v_id_venta       NUMBER;
BEGIN
    -- Validar canal de cobro
    IF UPPER(p_metodo_pago) NOT IN ('EFECTIVO', 'YAPE', 'PLIN', 'TRANSFERENCIA') THEN
        RAISE_APPLICATION_ERROR(-20011, 'Método de pago no válido.');
    END IF;

    -- Bloqueo pesimista concurrente
    SELECT STOCK, PRECIO, NOMBRE 
    INTO v_stock_actual, v_precio_prod, v_nombre_prod
    FROM PRODUCTOS 
    WHERE ID = p_id_producto 
    FOR UPDATE;

    -- Validación RN-VNT-01
    IF v_stock_actual < p_cantidad THEN
        RAISE_APPLICATION_ERROR(-20010, 
            'Regla RN-VNT-01: Stock insuficiente para ' || v_nombre_prod || 
            '. Disponible: ' || v_stock_actual || ' unidades.');
    END IF;

    -- Cálculos comerciales
    v_subtotal_item := v_precio_prod * p_cantidad;
    v_igv := ROUND(v_subtotal_item * 0.18, 2);
    v_base_neta := v_subtotal_item - v_igv;
    p_numero_ticket := 'TKT-' || TO_CHAR(SYSDATE, 'YYYYMMDDHH24MISS');
    p_total_cobrado := v_subtotal_item;

    -- Inserción en cabecera
    INSERT INTO VENTAS (NUMERO_TICKET, FECHA, ESTADO, METODO_PAGO, SUBTOTAL, IGV, TOTAL, CLIENTE_NOMBRE)
    VALUES (p_numero_ticket, SYSDATE, 'REGISTRADA', UPPER(p_metodo_pago), v_base_neta, v_igv, v_subtotal_item, p_cliente_nombre)
    RETURNING ID INTO v_id_venta;

    -- Inserción en detalle
    INSERT INTO DETALLE_VENTAS (ID_VENTA, ID_PRODUCTO, ID_LOTE, NOMBRE_PRODUCTO, PRECIO_UNITARIO, CANTIDAD, SUBTOTAL)
    VALUES (v_id_venta, p_id_producto, p_id_lote, v_nombre_prod, v_precio_prod, p_cantidad, v_subtotal_item);

    -- Actualización de inventario físico
    UPDATE PRODUCTOS 
    SET STOCK = STOCK - p_cantidad
    WHERE ID = p_id_producto;

    IF p_id_lote IS NOT NULL THEN
        UPDATE LOTES_PRODUCTO
        SET STOCK_DISPONIBLE = STOCK_DISPONIBLE - p_cantidad
        WHERE ID = p_id_lote;
    END IF;

    COMMIT;
EXCEPTION
    WHEN OTHERS THEN
        ROLLBACK;
        RAISE;
END;

4.2 Stored Procedure 02: Anulación de Venta con Reposición Automática de Inventario

CREATE OR REPLACE PROCEDURE SP_ANULAR_VENTA (
    p_id_venta IN NUMBER
) AS
    v_estado_actual VARCHAR2(20);
BEGIN
    SELECT ESTADO 
    INTO v_estado_actual
    FROM VENTAS 
    WHERE ID = p_id_venta 
    FOR UPDATE;

    IF v_estado_actual = 'ANULADA' THEN
        RAISE_APPLICATION_ERROR(-20015, 'La venta ya se encuentra anulada previamente.');
    END IF;

    -- Devolución de existencias al almacén
    FOR item IN (SELECT ID_PRODUCTO, ID_LOTE, CANTIDAD FROM DETALLE_VENTAS WHERE ID_VENTA = p_id_venta) LOOP
        UPDATE PRODUCTOS 
        SET STOCK = STOCK + item.CANTIDAD 
        WHERE ID = item.ID_PRODUCTO;

        IF item.ID_LOTE IS NOT NULL THEN
            UPDATE LOTES_PRODUCTO 
            SET STOCK_DISPONIBLE = STOCK_DISPONIBLE + item.CANTIDAD 
            WHERE ID = item.ID_LOTE;
        END IF;
    END LOOP;

    -- Asentar estado anulado
    UPDATE VENTAS 
    SET ESTADO = 'ANULADA' 
    WHERE ID = p_id_venta;

    COMMIT;
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        ROLLBACK;
        RAISE_APPLICATION_ERROR(-20016, 'No existe la venta indicada.');
    WHEN OTHERS THEN
        ROLLBACK;
        RAISE;
END;
/

5.1 Función 01: Determinar Lote Prioritario por Vencimiento (Criterio FEFO)

Retorna el CODIGO_LOTE que debe salir primero del almacén para garantizar que el producto no caduque en anaquel:

CREATE OR REPLACE FUNCTION FN_OBTENER_LOTE_FEFO (
    p_id_producto IN NUMBER,
    p_cantidad    IN NUMBER
) RETURN VARCHAR2 AS
    v_codigo_lote VARCHAR2(30);
BEGIN
    SELECT CODIGO_LOTE
    INTO v_codigo_lote
    FROM (
        SELECT CODIGO_LOTE
        FROM LOTES_PRODUCTO
        WHERE ID_PRODUCTO = p_id_producto
          AND STOCK_DISPONIBLE >= p_cantidad
        ORDER BY FECHA_VENCIMIENTO ASC
    )
    WHERE ROWNUM = 1;

    RETURN v_codigo_lote;
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        RETURN 'SIN_LOTE_DISPONIBLE';
END;
/

Prueba de la Función 01:

SELECT 
    p.NOMBRE AS PRODUCTO,
    FN_OBTENER_LOTE_FEFO(p.ID, 10) AS LOTE_PRIORITARIO_FEFO
FROM PRODUCTOS p
WHERE p.ID IN (1, 6);

5.2 Función 02: Reporte de Total Recaudado por Método de Pago

Totaliza el volumen de facturación en soles acumulado para un determinado método de pago en ventas no anuladas:

CREATE OR REPLACE FUNCTION FN_TOTAL_VENTAS_METODO (
    p_metodo_pago IN VARCHAR2
) RETURN NUMBER AS
    v_total NUMBER(12, 2) := 0;
BEGIN
    SELECT NVL(SUM(TOTAL), 0)
    INTO v_total
    FROM VENTAS
    WHERE UPPER(METODO_PAGO) = UPPER(p_metodo_pago)
      AND ESTADO = 'REGISTRADA';

    RETURN v_total;
END;
/

Prueba de la Función 02:

    FN_TOTAL_VENTAS_METODO('EFECTIVO') AS RECAUDADO_EFECTIVO,
    FN_TOTAL_VENTAS_METODO('YAPE') AS RECAUDADO_YAPE,
    FN_TOTAL_VENTAS_METODO('TRANSFERENCIA') AS RECAUDADO_TRANSFERENCIA
FROM DUAL;
